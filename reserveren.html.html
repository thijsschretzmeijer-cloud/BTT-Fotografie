<script>

  /*

   * GOOGLE APPS SCRIPT URL

   */

  const GOOGLE_SCRIPT_URL =

    "https://script.google.com/macros/s/AKfycbwA-BnrYGMeUoQQsBSHoxTkPpPeYCB1WsQGIe9FaF5OWsoRwNuJxtAZHZp4Y9Fm8CU/exec";

  /*

   * BESCHIKBARE TIJDEN

   */

  const tijden = [

    "09:30 – 10:25",

    "11:15 – 12:10",

    "13:00 – 13:55",

    "14:30 – 15:25",

    "16:00 – 16:55"

  ];

  let gekozenDatum = "";

  let gekozenTijd = "";

  /*

   * BESCHIKBAARHEID OPHALEN

   */

  async function laadBeschikbaarheid() {

    gekozenDatum =

      document.getElementById("datum").value;

    gekozenTijd = "";

    document.getElementById(

      "gekozenAfspraak"

    ).textContent = "Nog niet gekozen";

    const container =

      document.getElementById("tijdsloten");

    container.innerHTML =

      "<div>Beschikbaarheid wordt geladen...</div>";

    if (!gekozenDatum) {

      container.innerHTML =

        "<div>Kies eerst een datum.</div>";

      return;

    }

    try {

      const response = await fetch(

        GOOGLE_SCRIPT_URL

      );

      const gegevens = await response.json();

      /*

       * De Google Sheet geeft de gereserveerde

       * afspraken terug.

       */

      const reserveringen =

        gegevens.reserveringen || [];

      container.innerHTML = "";

      tijden.forEach(function(tijd) {

        const knop =

          document.createElement("button");

        knop.type = "button";

        knop.className = "tijdslot";

        const bezet =

          reserveringen.some(function(reservering) {

            return (

              reservering.datum === gekozenDatum &&

              reservering.tijd === tijd

            );

          });

        if (bezet) {

          knop.classList.add("bezet");

          knop.disabled = true;

          knop.textContent =

            tijd + " – BEZET";

        } else {

          knop.textContent = tijd;

          knop.onclick = function() {

            kiesTijd(tijd, knop);

          };

        }

        container.appendChild(knop);

      });

    } catch (error) {

      console.error(error);

      container.innerHTML =

        "<div>De beschikbaarheid kon niet worden geladen. Probeer het opnieuw.</div>";

    }

  }

  /*

   * TIJD KIEZEN

   */

  function kiesTijd(tijd, knop) {

    document

      .querySelectorAll(".tijdslot")

      .forEach(function(item) {

        item.classList.remove(

          "geselecteerd"

        );

      });

    knop.classList.add(

      "geselecteerd"

    );

    gekozenTijd = tijd;

    document.getElementById(

      "gekozenAfspraak"

    ).textContent =

      gekozenDatum +

      " | " +

      gekozenTijd;

  }

  /*

   * RESERVERING PLAATSEN

   */

  async function reserveren() {

    const naam =

      document

        .getElementById("naam")

        .value

        .trim();

    const email =

      document

        .getElementById("email")

        .value

        .trim();

    const telefoon =

      document

        .getElementById("telefoon")

        .value

        .trim();

    const melding =

      document.getElementById(

        "melding"

      );

    const knop =

      document.getElementById(

        "reserverenKnop"

      );

    melding.style.display =

      "block";

    /*

     * CONTROLES

     */

    if (!gekozenDatum || !gekozenTijd) {

      melding.textContent =

        "Kies eerst een datum en een tijdstip.";

      return;

    }

    if (!naam || !email || !telefoon) {

      melding.textContent =

        "Vul alstublieft uw naam, e-mailadres en telefoonnummer in.";

      return;

    }

    /*

     * KNOP UITSCHAKELEN

     */

    knop.disabled = true;

    knop.textContent =

      "RESERVERING WORDT VERWERKT...";

    try {

      const response =

        await fetch(

          GOOGLE_SCRIPT_URL,

          {

            method: "POST",

            body: JSON.stringify({

              datum: gekozenDatum,

              tijd: gekozenTijd,

              naam: naam,

              email: email,

              telefoon: telefoon

            })

          }

        );

      const resultaat =

        await response.json();

      /*

       * SLOT IS AL BEZET

       */

      if (

        resultaat.status ===

        "bezet"

      ) {

        melding.textContent =

          "Dit tijdslot is helaas zojuist door iemand anders gereserveerd. Kies een ander tijdstip.";

        knop.disabled = false;

        knop.textContent =

          "RESERVERING PLAATSEN";

        laadBeschikbaarheid();

        return;

      }

      /*

       * SUCCES

       */

      if (

        resultaat.status ===

        "success"

      ) {

        melding.innerHTML =

          "Bedankt " +

          naam +

          "!<br><br>" +

          "Uw reservering is succesvol geplaatst.<br><br>" +

          "<strong>" +

          gekozenDatum +

          " | " +

          gekozenTijd +

          "</strong>";

        /*

         * Velden leegmaken

         */

        document.getElementById(

          "naam"

        ).value = "";

        document.getElementById(

          "email"

        ).value = "";

        document.getElementById(

          "telefoon"

        ).value = "";

        /*

         * Opnieuw beschikbaarheid ophalen

         */

        laadBeschikbaarheid();

        knop.disabled = false;

        knop.textContent =

          "RESERVERING PLAATSEN";

        return;

      }

      /*

       * FOUTMELDING

       */

      melding.textContent =

        resultaat.bericht ||

        "Er is iets misgegaan.";

      knop.disabled = false;

      knop.textContent =

        "RESERVERING PLAATSEN";

    } catch (error) {

      console.error(error);

      melding.textContent =

        "Er is een technische fout opgetreden. Controleer de koppeling met Google Apps Script.";

      knop.disabled = false;

      knop.textContent =

        "RESERVERING PLAATSEN";

    }

  }

</script>
